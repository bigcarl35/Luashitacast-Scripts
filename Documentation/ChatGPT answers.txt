function IsShieldEquipped()
    -- Get the inventory memory manager
    local inventory = AshitaCore:GetMemoryManager():GetInventory();
    if not inventory then return false end

    -- Fetch the item index currently in the Sub-Weapon slot (Slot 1)
    local equipSlot = inventory:GetEquippedItem(1);
    if not equipSlot or equipSlot.ItemIndex == 0 then
        return false -- Nothing equipped in the sub slot
    end

    -- Convert the equipment index to the actual item object from the correct bag
    local item = inventory:GetContainerItem(equipSlot.ContainerIndex, equipSlot.ItemIndex);
    if not item or item.Id == 0 then return false end

    -- Retrieve the static resource data for this item ID
    local resourceItem = AshitaCore:GetResourceManager():GetItemById(item.Id);
    if not resourceItem then return false end

    -- Check if the item type belongs to a shield (Type flag 4)
    return resourceItem.Type == 4
end

Is there any way to track cooldowns on charged items?

Not specifically, but you can register when an item is used and have and check an internal timer. This will not
work if you change jobs since the portion that tracks that an event will be reloaded, but within a job this is
possible.

I asked specifically about warp cudgels, but if you have a dynamic table that tracks item/storage location and
a start/duration value, you can determine when the item is valid to use.

local warpCudgelCooldown = 600 -- Cooldown in seconds
local lastWarpCudgelUse = 0    -- Timestamp of last use

--

profile.HandleCommand = function(args)
    -- Create a custom command like /addon lac warp
    if (args[1] == 'warp') then
        local currentTime = os.clock()
        local timeElapsed = currentTime - lastWarpCudgelUse

        -- Check if item is still on cooldown
        if (timeElapsed < warpCudgelCooldown) then
            local remaining = math.ceil(warpCudgelCooldown - timeElapsed)
            AshitaCore:GetChatManager():Write('Warp Cudgel is on cooldown! ' .. remaining .. 's remaining.')
            return true -- Blocks the command
        end

        -- If not on cooldown, equip and use it
        -- Note: FXI items require a delay to use after equipping
        AshitaCore:GetChatManager():Write('Equipping Warp Cudgel...')
        AshitaCore:GetInputManager():SendCommand('/equip main "Warp Cudgel"', 4)

        -- Use the item after a 4-second delay, then re-equip previous weapon
        local useCommand = '/wait 4; /item "Warp Cudgel" <me>; /wait 4; /lac enable main;'
        AshitaCore:GetInputManager():SendCommand(useCommand, 4)

        -- Lock the slot so LuAshitacast doesn't swap it out immediately
        AshitaCore:GetInputManager():SendCommand('/lac disable main', 4)

        -- Update the cooldown timer
        lastWarpCudgelUse = currentTime
        return true
    end

    return false;
end;

local function find_warp_cudgel_container()
    -- Warp Cudgel item ID is 17040

    -- List of common accessible containers:
    -- 0: Inventory, 8: Wardrobe 1, 10-17: Wardrobes 2-8

    end

    return nil, "Not found in accessible bags"

Key Concepts to Know for Ashita 4 ImGui

The Reference Table Design Pattern ({ value }): ImGui is originally written in C++, which uses memory pointers to mutate data inside an argument list. Because standard Lua
variables cannot be passed by reference, Ashita wraps variables inside a single-element table array. Notice how the open state is checked via isWindowOpen[1] and passed
directly into imgui.begin as a table container.

Execution Frequency: The render function runs roughly 30 to 60+ times per second depending on your game's framerate. Never execute slow operations inside this loop (like
looping through entire inventory bags or searching for files). Keep it strictly focused on visual generation.

imgui.begin and imgui.end Rules: Every open tag (imgui.begin()) must eventually be closed by a matching tag (imgui.end()) within the same frame block, or your game client
will instantly crash to your desktop.

can you display images using offsets in imgui?

How It Works (UV Coordinates)ImGui maps image selections using normalized float values ranging from 0.0 to 1.0:
(0.0, 0.0) represents the top-left corner of your texture file.
(1.0, 1.0) represents the bottom-right corner of your texture file.

By calculating the pixel offset and dividing it by the total dimensions of the texture file, you can pinpoint the exact sub-image you want to crop.

local imgui = require('imgui')

-- Assume you loaded your texture via Ashita's asset manager/renderer system
-- (e.g., using ASSET_ID or your texture pointer)
local mySpritesheetTexture = get_your_texture_pointer_here()

-- Spritesheet details
local sheetWidth = 256
local sheetHeight = 256
local iconSize = 32

-- The specific frame/icon offset we want to crop (Grid row and column)
local iconColumn = 2  -- 3rd column (0-indexed: 0, 1, 2)
local iconRow = 1     -- 2nd row (0-indexed: 0, 1)

-- Calculate pixel boundaries within the sheet
local pixelX_Min = iconColumn * iconSize
local pixelY_Min = iconRow * iconSize
local pixelX_Max = pixelX_Min + iconSize
local pixelY_Max = pixelY_Min + iconSize

-- Convert pixel offsets into ImGui normalized UV coordinates (0.0 to 1.0)
local uv0_X = pixelX_Min / sheetWidth
local uv0_Y = pixelY_Min / sheetHeight
local uv1_X = pixelX_Max / sheetWidth
local uv1_Y = pixelY_Max / sheetHeight

ashita.events.register('render', 'sprite_render_event', function()
    if imgui.begin('Icon Offset Window', table_hold, ImGuiWindowFlags_AlwaysAutoResize) then

        imgui.text('Displaying cropped icon using UV offsets:')

        -- Ashita 4 ImGui API signature requires raw floats instead of vectors:
        -- imgui.image(textureID, sizeX, sizeY, uv0_X, uv0_Y, uv1_X, uv1_Y)
        imgui.image(
            mySpritesheetTexture,
            iconSize, iconSize,     -- The display dimensions in your UI window
            uv0_X, uv0_Y,           -- Top-Left UV boundary
            uv1_X, uv1_Y            -- Bottom-Right UV boundary
        )

        imgui.end()
    end
end)


Visual Shifts: Canvas Offsets vs. Texture Offsets

Depending on what you mean by "offset," you have two distinct approaches in your UI design:

    Texture Offset (Cropping / Spritesheets): Done via the code above using uv0 and uv1 coordinates to selectively display a portion of an image.

    Canvas Offset (Position Shifting): If you want to draw an image at a specific absolute pixel coordinate or offset relative to the window, you
    should use ImDrawList. You fetch the draw list using imgui.getWindowDrawList() and draw directly via drawList:addImage() using absolute
    screenspace limits.

I have a series of the same image with different highlights. I want to show the right slice based on what day it is in ffxi

Final Fantasy XI cycles through its 8 days in a fixed, permanent sequence (Firesday to Lightsday). The game's memory manager assigns each day an index
from 0 to 7.

The 8 Vanadiel Days Index Map
    0: Firesday
    1: Earthsday
    2: Watersday
    3: Windsday
    4: Icesday
    5: Lightningday
    6: Lightsday
    7: Darksday

local imgui = require('imgui')

-- Setup texture metadata
local texturePointer = get_your_texture_pointer() -- Replace with your texture asset pointer
local sliceWidth = 32   -- Width of one day's highlight slice
local sliceHeight = 32  -- Height of the image slice
local totalWidth = 256  -- Total width of your combined image (32 * 8)
local totalHeight = 32  -- Total height of your image

ashita.events.register('render', 'ffxi_day_image_render', function()
    if imgui.begin('Vana\'diel Day Highlight', table_hold, ImGuiWindowFlags_AlwaysAutoResize) then

        -- 1. Get the current FFXI day index (0 to 7) from memory
        local timeManager = AshitaCore:GetMemoryManager():GetTime()
        local currentDayIndex = 0

        if timeManager ~= nil then
            currentDayIndex = timeManager:GetDay() -- Returns 0 for Firesday, 1 for Earthsday, etc.
        end

        -- 2. Calculate the pixel boundaries for today's slice
        -- Example: Earthsday (Index 1) -> Min X = 32, Max X = 64
        local pixelX_Min = currentDayIndex * sliceWidth
        local pixelX_Max = pixelX_Min + sliceWidth

        -- 3. Convert pixel boundaries into normalized UV coordinates (0.0 to 1.0)
        local uv0_X = pixelX_Min / totalWidth
        local uv0_Y = 0.0 -- Top of the texture
        local uv1_X = pixelX_Max / totalWidth
        local uv1_Y = 1.0 -- Bottom of the texture

        imgui.text(string.format("Today's Day Index: %d", currentDayIndex))
        imgui.separator()

        -- 4. Render the specific highlighted slice
        imgui.image(
            texturePointer,
            sliceWidth, sliceHeight, -- Visual size inside the ImGui frame
            uv0_X, uv0_Y,            -- Top-Left UV coordinate
            uv1_X, uv1_Y             -- Bottom-Right UV coordinate
        )

        imgui.end()
    end
end)

Layout Variations

If your image file is arranged differently, you only need to change how pixelX_Min and pixelY_Min are calculated:

Arranged Vertically (Stacked): Set pixelX_Min = 0 and change your Y bounds to scale with the day:
local pixelY_Min = currentDayIndex * sliceHeight.

Arranged as a Grid (e.g., 4x2 grid): Use modulo and division to get the row and column:

local columns = 4
local col = currentDayIndex % columns
local row = math.floor(currentDayIndex / columns)
local pixelX_Min = col * sliceWidth
local pixelY_Min = row * sliceHeight
